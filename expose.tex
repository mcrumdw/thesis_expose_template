\section{Introduction and Motivation}

\guidance{%
    \textbf{Purpose of this section:} Introduce the topic to a reader who is
    knowledgeable in the field but unfamiliar with your specific problem.
    Aim for 1--2 pages.\\[0.5em]
    \textbf{What to include:}
    \begin{itemize}
        \item \textbf{Context \& relevance} — Why does this topic matter today?
              Mention industry trends, societal challenges, or scientific gaps
              that make this research timely.
        \item \textbf{Problem hook} — Describe a concrete situation or deficiency
              that your work will address.  Use statistics or a recent event if
              available.
        \item \textbf{Scope statement} — Briefly state what the thesis will and
              will \emph{not} cover.
        \item \textbf{Structure overview} — One sentence per subsequent section
              explaining how the exposé is organised.
    \end{itemize}
    \textbf{Tip:} Write this section last, once all other sections are complete.%
}

% ══════════════════════════════════════════════════════════
%  2  PROBLEM STATEMENT & RESEARCH QUESTIONS
% ══════════════════════════════════════════════════════════
\section{Problem Statement and Research Questions}

\guidance{%
    \textbf{Purpose of this section:} Precisely articulate the gap, challenge,
    or open question your thesis addresses.  Aim for 0.5--1 page.\\[0.5em]
    \textbf{What to include:}
    \begin{itemize}
        \item A concise statement (2--4 sentences) of the core problem.
        \item An explanation of \emph{why} existing solutions are insufficient.
        \item \textbf{Research questions (RQs)} — formulate 1--3 specific,
              answerable questions.  Each RQ should be:
              \begin{itemize}
                  \item \emph{Specific}: addresses one concept at a time.
                  \item \emph{Feasible}: answerable within the thesis timeline.
                  \item \emph{Relevant}: directly linked to the problem.
              \end{itemize}
        \item Optionally, a research \textbf{hypothesis} if you have one.
    \end{itemize}
    \textbf{Tip:} Your RQs are the anchor for every other section — methodology,
    evaluation criteria, and conclusions all flow from them.%
}

% ══════════════════════════════════════════════════════════
%  3  OBJECTIVES
% ══════════════════════════════════════════════════════════
\section{Objectives}

\guidance{%
    \textbf{Purpose of this section:} Translate your research questions into
    concrete, measurable goals.  Aim for 0.5 pages.\\[0.5em]
    \textbf{What to include:}
    \begin{itemize}
        \item \textbf{Primary objective} — the single most important goal (maps
              directly to your main RQ).
        \item \textbf{Secondary objectives} — supporting goals (e.g., design an
              artefact, conduct a user study, develop a framework).
        \item Each objective should start with an action verb: \emph{design,
              implement, evaluate, analyse, compare, develop, validate, \ldots}
    \end{itemize}
    \textbf{Tip:} Objectives are \emph{not} research questions.  An objective
    describes what you will \emph{do}; an RQ asks what you want to
    \emph{find out}.%
}

%  4  RELATED WORK

\section{Related Work}

\guidance{%
    \textbf{Purpose of this section:} Demonstrate awareness of existing
    literature and position your work relative to it.  Aim for 1--2 pages.\\[0.5em]
    \textbf{What to include:}
    \begin{itemize}
        \item \textbf{Search strategy} (brief) — which databases, keywords, and
              date range you used (e.g., ACM Digital Library, IEEE Xplore,
              Google Scholar; last 10 years).
        \item \textbf{Thematic clusters} — group related papers by sub-topic,
              not by chronology.  Use one subsection per cluster.
        \item \textbf{Critical synthesis} — do not merely summarise each paper;
              compare, contrast, and identify common limitations.
        \item \textbf{Gap identification} — conclude with the specific gap your
              thesis fills, explicitly referencing your RQs.
    \end{itemize}
    \textbf{Tip:} Every claim should be backed by a citation \cite{doe_example_2024}.
    Aim for at least 10--15 sources in the exposé.%
}

% ══════════════════════════════════════════════════════════
%  5  METHODOLOGY
% ══════════════════════════════════════════════════════════
\section{Methodology}

\guidance{%
    \textbf{Purpose of this section:} Describe \emph{how} you will answer your
    research questions.  This is often the most critical section for
    supervisors.  Aim for 1--2 pages.\\[0.5em]
    \textbf{What to include:}
    \begin{itemize}
        \item \textbf{Research design} — qualitative, quantitative, or mixed?
              Empirical study, design science, systematic review, simulation, \ldots?
        \item \textbf{Use case / study context} — the specific system, domain, or
              dataset you will work with.
        \item \textbf{Variables or evaluation criteria} — what you measure and
              how (independent, dependent, control variables for experiments;
              or evaluation criteria for design artefacts).
        \item \textbf{Data collection} — surveys, interviews, logs, benchmarks,
              public datasets, \ldots
        \item \textbf{Analysis methods} — statistical tests, qualitative coding,
              performance benchmarks, \ldots
        \item \textbf{Validity considerations} — how you will ensure internal
              and external validity, reliability, and ethical compliance.
    \end{itemize}
    \textbf{Tip:} Justify every methodological choice by linking it back to
    your RQs.%
}

\subsection{Use Case Definition}

\guidance{Describe the application domain, system, or dataset that will serve
as the object of study.  Explain why this use case is appropriate for
answering your RQs.  Include any relevant constraints (access, size,
language, \ldots).}

\subsection{Independent Variables}

\guidance{List and define the variables (or design parameters) that you will
deliberately manipulate or vary.  For each variable, state its possible
values or levels.}

\subsection{Dependent Variables}

\guidance{List and define the outcomes you will measure.  For each dependent
variable, specify the measurement instrument or metric (e.g., task completion
time in seconds, SUS score 0--100, F1-score, \ldots).}

\subsection{Procedure / Research Steps}

\guidance{Provide a step-by-step overview of how the study or development work
will be carried out — from setup and data collection through to analysis.
A numbered list or small diagram works well here.}

% ══════════════════════════════════════════════════════════
%  6  EXPECTED CONTRIBUTIONS
% ══════════════════════════════════════════════════════════
\section{Expected Contributions and Results}

\guidance{%
    \textbf{Purpose of this section:} Communicate the anticipated value of your
    thesis.  Aim for 0.5--1 page.\\[0.5em]
    \textbf{What to include:}
    \begin{itemize}
        \item \textbf{Scientific contribution} — new knowledge, a validated
              model, a benchmark, an empirically tested framework, \ldots
        \item \textbf{Practical / applied contribution} — a prototype, tool,
              guideline, or design recommendation.
        \item \textbf{Expected findings} — your educated hypothesis about what
              you will discover (be honest about uncertainty; phrases like
              \emph{``we expect that \ldots''} or \emph{``it is hypothesised
              that \ldots''} are appropriate).
        \item \textbf{Limitations} — acknowledge foreseeable constraints
              (sample size, generalisation, time, \ldots).
    \end{itemize}
    \textbf{Tip:} Do not over-promise.  Reviewers value intellectual honesty
    over grand claims.%
}
